### 1.2 Inspect a sample weekly rollup for the `comments` sub-object

```bash
# Pick the most recent rollup
python -c "
import json, glob, pathlib
files = sorted(glob.glob('docs/data/aggregates/weekly_rollups/*.json'))
data = json.loads(pathlib.Path(files[-1]).read_text())
print('week:', data['week'])
print('pr_count:', data['pr_count'])
print('comments object present:', 'comments' in data)
if 'comments' in data:
    c = data['comments']
    print('comments fields:', sorted(c.keys()))
    print('thread_count:', c['thread_count'])
    print('comment_count:', c['comment_count'])
    print('active_thread_count:', c['active_thread_count'])
    print('coverage_partial:', c['coverage_partial'])
    print('atomicity OK:', set(c.keys()) == {'thread_count', 'comment_count', 'active_thread_count', 'coverage_partial'})
    print('INV-1-06 active <= thread:', c['active_thread_count'] <= c['thread_count'])
"
# Expected (capability-on): comments object present == True; all four fields enumerated;
# atomicity OK == True; INV-1-06 holds.
```
